FastAPI Agent Endpoint with Tracing + Retry

from fastapi import FastAPI, HTTPException
from langfuse import Langfuse
from tenacity import retry, wait_exponential, stop_after_attempt
import logging

app  = FastAPI(); lf = Langfuse()
log  = logging.getLogger("agent")

@retry(wait=wait_exponential(min=1, max=10),
       stop=stop_after_attempt(3))
async def llm_call(messages):
    return await openai_client.chat.completions.create(
        model="gpt-4o-mini", messages=messages)

@app.post("/agent/run")
async def run_agent(req):
    trace = lf.trace(name="agent", user_id=req.user_id, input=req.query)
    try:
        with trace.span(name="loop"):
            result = await agent.arun(req.query, user=req.user_id)
        trace.update(output=result.text, usage=result.usage)
        log.info("agent_done", extra={"tokens": result.usage})
        return {"answer": result.text, "tool_calls": result.tool_calls}
    except Exception as e:
        trace.update(error=str(e)); log.exception("agent_failed")
        raise HTTPException(500, "agent error")
